# Install as .github/workflows/oracle.yml in the own-firmware-oracle repo.
# Shipped as a template (not an active workflow). This one touches firmware,
# so read the R2 / environment notes below before enabling it.
#
# Secrets live ONLY in a protected `firmware` environment limited to `main`.
# Fork PRs never see them. No pull_request_target. Firmware is fetched to a
# cache OUTSIDE the repo and deleted after the run; never cached, never an
# artifact (a fork PR could otherwise restore it from the base-branch cache).
name: oracle
on:
  push:
    branches: [main]
  workflow_dispatch:
  schedule:
    - cron: "0 4 * * 1"   # weekly
permissions:
  contents: write         # the final job opens a PR with changed TSVs
  pull-requests: write
jobs:
  plan:
    runs-on: ubuntu-latest
    outputs:
      matrix: ${{ steps.plan.outputs.matrix }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - run: pip install pyyaml
      # Emit the catalog entries whose (image sha256, tool version) key is not
      # yet reflected in results/. Only out-of-date images get rebuilt.
      - id: plan
        run: python tools/plan.py --emit-matrix >> "$GITHUB_OUTPUT"

  oracle:
    needs: plan
    if: ${{ needs.plan.outputs.matrix != '[]' }}
    runs-on: ubuntu-latest
    environment: firmware        # protected; only main can read the secrets
    strategy:
      max-parallel: 4
      matrix:
        entry: ${{ fromJson(needs.plan.outputs.matrix) }}
    env:
      R2_ENDPOINT: ${{ secrets.R2_ENDPOINT }}
      R2_ACCESS_KEY_ID: ${{ secrets.R2_ACCESS_KEY_ID }}
      R2_SECRET_ACCESS_KEY: ${{ secrets.R2_SECRET_ACCESS_KEY }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - run: pip install pyyaml boto3
      - run: sudo apt-get update && sudo apt-get install -y e2fsprogs
      - name: fetch (verified by sha256, cache is outside the repo)
        run: echo "IMG=$(python tools/fwfetch.py ${{ matrix.entry }})" >> "$GITHUB_ENV"
      - name: unpack -> manifest
        run: |
          out="results/$(python tools/plan.py --result-path ${{ matrix.entry }})"
          python tools/unpack.py ${{ matrix.entry }} "$IMG" -o "$out"
      - name: delete firmware from the runner
        if: always()
        run: rm -rf ~/.cache/own-fw
      - name: guard
        run: python tools/guard.py
      - uses: peter-evans/create-pull-request@v6
        with:
          branch: oracle/update-results
          title: "oracle: refresh firmware manifests"
          commit-message: "oracle: refresh firmware manifests"
